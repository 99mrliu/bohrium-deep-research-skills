%% Compile:  pdflatex example-chart.tex

\documentclass[tikz,border=6pt]{standalone}
\input{biopathway.tex}

\begin{document}
\begin{tikzpicture}
\begin{axis}[
    bpchart,
    xlabel={$\log_2$ fold change},
    ylabel={$-\log_{10}(p)$},
    title={Synthetic volcano-style demo (example-data.csv)},
    legend pos=north west,
  ]
  %% up-regulated points (log2FC > 0): filter via y expr is omitted for
  %% minimalism; a single scatter series keeps the example dependency-free.
  \addplot[bpscatter] table[x=log2FC, y=neglog10p, col sep=comma]
    {example-data.csv};
  \addlegendentry{genes (synthetic)}
  %% significance guide lines
  \draw[dashed, bpgray] (axis cs:1,-1) -- (axis cs:1,11);
  \draw[dashed, bpgray] (axis cs:-1,-1) -- (axis cs:-1,11);
  \draw[dashed, bpgray] (axis cs:-4,2) -- (axis cs:4,2);
\end{axis}
\end{tikzpicture}
\end{document}
